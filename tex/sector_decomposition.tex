\section{Setting and overview}

\subsection{The integral}

Let $\ell_1,\dots,\ell_d\ge0$ and, for $s>0$, let
$A_s=\operatorname{diag}(s^{\ell_1},\dots,s^{\ell_d})$. For $t\ge1$ and $\xi\in\R^d$
we define the phase
\[
\psi(t,\xi)=\int_1^t|A_s\xi|\,ds .
\]
We study the oscillatory integral
\[
I(t,x)=\int_{\R^d}e^{i(x\cdot\xi-\psi(t,\xi))}\,|A_t\xi|^{-1}\,\chi(\xi)\,d\xi ,
\]
where $\chi$ is a smooth cutoff supported in the annulus $\{\tfrac12\le|\xi|\le2\}$.
We want to bound $I(t,x)$ for large $t$, uniformly in $x$.

\subsection{The difficulty and the plan}

The matrix $A_t$ stretches each frequency coordinate by a different power of $t$.
So there is no single large parameter in front of the phase. We handle this in two
moves.

First we split frequency space into $d$ sectors, according to which weighted
coordinate $t^{\ell_j}|\xi_j|$ is the largest. On sector $k$ we use the dominant
coordinate $\xi_k$ and the $d-1$ ratios $\eta_j=t^{\ell_j}\xi_j/(t^{\ell_k}\xi_k)$ as
new variables. In these variables the sector integral becomes an outer integral in
$\xi_k$ of an inner oscillatory integral in $\eta$ with a single large parameter
$\lambda=t^{\ell_k+1}|\xi_k|$ (Section~\ref{sec:reduction}).

Second we bound the inner integral by stationary phase. The phase is strictly
concave in $\eta$ (Section~\ref{sec:hessian}). Its higher derivatives are not
bounded uniformly in $t$, but they are once we weight them correctly
(Section~\ref{sec:higher}). A direct application of the stationary phase theorem
gives decay in $\lambda$ but loses a power of $t$ through the amplitude
(Section~\ref{sec:spt}). The loss disappears when the estimate is carried out at the
natural scale $\lambda^{-1/2}$. This gives the main result, Theorem~\ref{thm:main}
(Section~\ref{sec:proof}). The outer integral in $\xi_k$ is not treated here.

\subsection{Conventions}

In all estimates $t\ge2$. The index $k$ is fixed, and $i,j,m$ run over
$\{1,\dots,d\}\setminus\{k\}$. Multi-indices $\alpha=(\alpha_j)_{j\ne k}$ have $d-1$
entries, and $\delta_i$ is the multi-index with a $1$ in position $i$ and $0$
elsewhere. We write $C$, $c$ for positive constants that depend only on $d$ and
$\ell_1,\dots,\ell_d$ and may change from line to line. $C_\alpha$ may also depend
on $\alpha$.

\section{Reduction to one sector}\label{sec:reduction}

\subsection{Sectors}

Sector $k$ is the set of $\xi$ for which $t^{\ell_k}|\xi_k|$ is the largest of the
weights $t^{\ell_1}|\xi_1|,\dots,t^{\ell_d}|\xi_d|$. Since $|A_t\xi|^2$ is the sum
of the $d$ squared weights, on sector $k$
\[
t^{\ell_k}|\xi_k|\ \ge\ \frac{|A_t\xi|}{\sqrt d}.
\]
Every $\xi$ lies in some sector, so $I$ is a sum of $d$ sector integrals. We fix $k$.

\subsection{New variables}

For $j\ne k$ we introduce the ratios
\[
\eta_j=\frac{t^{\ell_j}\xi_j}{t^{\ell_k}\xi_k},
\]
which compare the weight of coordinate $j$ with the dominant weight. On sector $k$
they satisfy $|\eta_j|\le1$. We write $\eta=(\eta_j)_{j\ne k}\in\R^{d-1}$ and use
$(\xi_k,\eta)$ as coordinates on the sector.

\subsection{The integrand in the new variables}

We rewrite each factor of the integrand in terms of $\xi_k$ and $\eta$.

\paragraph{The amplitude.} Since
\[
|A_t\xi|^2=t^{2\ell_k}\xi_k^2+\sum_{j\ne k}t^{2\ell_j}\xi_j^2
=t^{2\ell_k}\xi_k^2\Big(1+\sum_{j\ne k}\eta_j^2\Big),
\]
we have
\[
|A_t\xi|=t^{\ell_k}|\xi_k|\,(1+|\eta|^2)^{1/2},
\qquad
|A_t\xi|^{-1}=t^{-\ell_k}|\xi_k|^{-1}(1+|\eta|^2)^{-1/2}.
\]

\paragraph{The phase $\psi$.} In the same way, for $0<s\le t$,
\[
|A_s\xi|=s^{\ell_k}|\xi_k|\Big(1+\sum_{j\ne k}\frac{s^{2\ell_j}\xi_j^2}{s^{2\ell_k}\xi_k^2}\Big)^{1/2}.
\]
We measure time in units of $t$: substitute $s=\sigma t$, so $ds=t\,d\sigma$ and
$s^{2(\ell_j-\ell_k)}\xi_j^2/\xi_k^2=\sigma^{2(\ell_j-\ell_k)}\eta_j^2$. This gives
\[
\psi(t,\xi)=t^{\ell_k+1}|\xi_k|\,G(t,\eta),
\qquad
G(t,\eta)=\int_{1/t}^1\sigma^{\ell_k}\Big(1+\sum_{j\ne k}\sigma^{2(\ell_j-\ell_k)}\eta_j^2\Big)^{1/2}d\sigma .
\]
So the phase is a large factor $t^{\ell_k+1}|\xi_k|$ times a function $G$ of
$\eta$, whose only dependence on $t$ is through the lower limit $1/t$.

\paragraph{The linear term.} We aim at a bound uniform in $x$, so we may replace $x$
by $A_tx$ without changing $\sup_x|I(t,x)|$. The linear part of the phase becomes
\[
A_tx\cdot\xi=x_kt^{\ell_k}\xi_k+\sum_{j\ne k}t^{\ell_j}x_j\xi_j
=x_kt^{\ell_k}\xi_k+t^{\ell_k}\xi_k\sum_{j\ne k}x_j\eta_j .
\]

\paragraph{The measure.} For fixed $\xi_k$, the change of variables
$\xi_j=t^{\ell_k-\ell_j}\xi_k\eta_j$, $j\ne k$, gives
\[
d\xi=|\xi_k|^{d-1}\prod_{j\ne k}t^{\ell_k-\ell_j}\;d\xi_k\,d\eta
=t^{\,d\ell_k-\sum_{j=1}^d\ell_j}\,|\xi_k|^{d-1}\,d\xi_k\,d\eta .
\]

\subsection{The sector integral}\label{sec:sector-integral}

Denote by $I_k$ the part of $I$ coming from sector $k$. Collecting the formulas
above,
\[
I_k=t^{\,d\ell_k-\sum_j\ell_j-\ell_k}\int_{\R}e^{ix_kt^{\ell_k}\xi_k}\,|\xi_k|^{d-2}\,
J(\xi_k)\,d\xi_k ,
\qquad
J(\xi_k)=\int_{[-1,1]^{d-1}}e^{i\lambda\Phi(\eta)}\,b(\xi_k,\eta)\,d\eta ,
\]
where the large parameter, the phase and the amplitude are
\[
\lambda=t^{\ell_k+1}|\xi_k| ,
\qquad
\Phi(\eta)=\frac{\xi_k}{|\xi_k|}\,\frac1t\sum_{j\ne k}x_j\eta_j-G(t,\eta),
\]
\[
b(\xi_k,\eta)=(1+|\eta|^2)^{-1/2}\,
\chi\Big(\xi_ke_k+\sum_{j\ne k}t^{\ell_k-\ell_j}\xi_k\eta_j\,e_j\Big).
\]
The rest of these notes bounds the inner integral $J$. Two remarks on its
ingredients.

\paragraph{The phase.} $\Phi$ depends on $x$ and on the sign of $\xi_k$ only
through its linear part. All derivatives of order two and higher are those of $-G$
and do not see $x$ at all.

\paragraph{The amplitude and the size of $\lambda$.} The second factor of $b$ is the
cutoff $\chi$ in the new variables. Since
$|\xi|=|\xi_k|\big(1+\sum_{j\ne k}t^{2(\ell_k-\ell_j)}\eta_j^2\big)^{1/2}$, the
amplitude $b$ is supported where
\[
\frac12\le|\xi_k|\Big(1+\sum_{j\ne k}t^{2(\ell_k-\ell_j)}\eta_j^2\Big)^{1/2}\le2 .
\]
With $\ell_{\min}=\min_j\ell_j$ and $|\eta_j|\le1$, this gives
\[
\frac{t^{\ell_{\min}-\ell_k}}{2\sqrt d}\ \le\ |\xi_k|\ \le\ 2
\qquad\text{on the support of } b,
\]
and therefore $\lambda\ge t^{1+\ell_{\min}}/(2\sqrt d)$. So $\lambda$ is indeed
large. Note, however, that a derivative $\partial_{\eta_j}$ falling on the cutoff
produces a factor $t^{\ell_k-\ell_j}\xi_k$, which is large when $\ell_j<\ell_k$.
This will be the main obstacle in Section~\ref{sec:spt}.

\section{The Hessian of the phase}\label{sec:hessian}

Set
\[
H(\sigma,\eta)=\Big(1+\sum_{j\ne k}\sigma^{2(\ell_j-\ell_k)}\eta_j^2\Big)^{1/2},
\qquad\text{so that}\qquad
G(t,\eta)=\int_{1/t}^1\sigma^{\ell_k}\,H(\sigma,\eta)\,d\sigma .
\]
The goal of this section is to show that $\Phi$ is strictly concave in $\eta$, with
a Hessian bounded below on compact sets uniformly in $t$.

\paragraph{First derivatives.} Since
$\partial_{\eta_j}H=\sigma^{2(\ell_j-\ell_k)}\eta_j\,H^{-1}$,
\[
\partial_{\eta_j}\Phi(\eta)=\frac{\xi_k}{|\xi_k|}\,\frac{x_j}{t}
-\eta_j\int_{1/t}^1\sigma^{2\ell_j-\ell_k}\,H^{-1}\,d\sigma .
\]

\paragraph{Second derivatives.} The linear part of $\Phi$ has zero Hessian, so
$\nabla_\eta^2\Phi=-\nabla_\eta^2G$. Differentiating once more,
\[
\partial_{\eta_i}\partial_{\eta_j}G(t,\eta)
=\int_{1/t}^1\sigma^{\ell_k}\,H^{-3}
\Big(\delta_{ij}\,\sigma^{2(\ell_j-\ell_k)}\,H^2
-\sigma^{2(\ell_i+\ell_j-2\ell_k)}\eta_i\eta_j\Big)\,d\sigma .
\]

\paragraph{The quadratic form.} Let
$D(\sigma)=\operatorname{diag}\big(\sigma^{2(\ell_j-\ell_k)}\big)_{j\ne k}$. In matrix
form the second derivatives read
\[
-\nabla_\eta^2\Phi
=\int_{1/t}^1\sigma^{\ell_k}H^{-3}\Big(H^2D-(D\eta)(D\eta)^{T}\Big)\,d\sigma .
\]
Fix $v\in\R^{d-1}$ and put $z=D^{1/2}\eta$, $u=D^{1/2}v$. Then
$\langle Dv,v\rangle=|u|^2$, $\langle D\eta,v\rangle=\langle z,u\rangle$, and
$|z|^2=\langle D\eta,\eta\rangle=H^2-1$. Hence
\[
-\langle\nabla_\eta^2\Phi\,v,v\rangle
=\int_{1/t}^1\sigma^{\ell_k}H^{-3}\Big(H^2|u|^2-\langle z,u\rangle^2\Big)\,d\sigma .
\]
By Cauchy--Schwarz, $\langle z,u\rangle^2\le|z|^2|u|^2=(H^2-1)|u|^2$, so the
bracket lies between $|u|^2$ and $H^2|u|^2$. Writing
\[
S(\sigma,v)=|u|^2=\sum_{j\ne k}\sigma^{2(\ell_j-\ell_k)}v_j^2 ,
\]
we arrive at the two-sided bound
\[
\int_{1/t}^1\sigma^{\ell_k}H^{-3}\,S\,d\sigma
\;\le\;-\langle\nabla_\eta^2\Phi\,v,v\rangle\;\le\;
\int_{1/t}^1\sigma^{\ell_k}H^{-1}\,S\,d\sigma .
\]

\paragraph{Strict concavity.} For $v\ne0$ the left side is positive. So
$-\nabla_\eta^2\Phi$ is positive definite and $\Phi$ is strictly concave in $\eta$,
for every $t$, $x$ and $\xi_k$.

\paragraph{A lower bound on compact sets.} Later we need a lower bound
$-\nabla_\eta^2\Phi\ge cI$ on a ball $|\eta|\le R$, with $c$ independent of $t$.
Since the integrand is nonnegative and $t\ge2$, we may restrict the integral to
$\sigma\in[\tfrac12,1]$. There $\sigma^{\ell_k}$ and each $\sigma^{2(\ell_j-\ell_k)}$
lie between two positive constants, so $S\ge c|v|^2$, and $H\le C_R$ when
$|\eta|\le R$. Hence
\[
-\langle\nabla_\eta^2\Phi(\eta)\,v,v\rangle
\ \ge\ \int_{1/2}^1\sigma^{\ell_k}H^{-3}\,S\,d\sigma\ \ge\ c_R\,|v|^2
\qquad(|\eta|\le R),
\]
with $c_R>0$ depending only on $R$, $d$ and the $\ell_j$.

\section{Higher derivatives of the phase}\label{sec:higher}

Let $|\alpha|\ge2$. Two derivatives kill the linear part of $\Phi$, so
$\partial_\eta^\alpha\Phi=-\partial_\eta^\alpha G$, and it suffices to bound
$\partial_\eta^\alpha G$. We first give a direct bound. Then we explain why it is
not uniform in $t$, and we prove a weighted bound that is.

\subsection{A direct bound}

\paragraph{Reduction to one function.} For fixed $\sigma$ put
$y_j=\sigma^{\ell_j-\ell_k}\eta_j$, $j\ne k$, and $\langle y\rangle=(1+|y|^2)^{1/2}$.
Then $H(\sigma,\eta)=\langle y\rangle$. Since each $y_j$ is a multiple of $\eta_j$
alone, the chain rule gives
\[
\partial_\eta^\alpha H
=\prod_{j\ne k}\sigma^{\alpha_j(\ell_j-\ell_k)}\;\big(\partial_y^\alpha\langle y\rangle\big).
\]

\paragraph{Derivatives of $\langle y\rangle$.} We claim that for $|\alpha|\ge1$,
$\partial_y^\alpha\langle y\rangle$ is a finite linear combination of terms
\[
y^\beta\,\langle y\rangle^{1-2m}\qquad\text{with}\qquad m\ge1,\quad2m-|\beta|=|\alpha| .
\]
For $|\alpha|=1$ this holds, since $\partial_{y_i}\langle y\rangle=y_i\langle y\rangle^{-1}$.
Differentiating $y^\beta\langle y\rangle^{1-2m}$ in $y_i$ gives two terms:
$\beta_i\,y^{\beta-\delta_i}\langle y\rangle^{1-2m}$, which vanishes if $\beta_i=0$,
and $(1-2m)\,y^{\beta+\delta_i}\langle y\rangle^{1-2(m+1)}$. In both terms
$2m-|\beta|$ increases by one, which proves the claim. Since
$|y^\beta|\le\langle y\rangle^{|\beta|}$, each term is bounded by
$\langle y\rangle^{|\beta|+1-2m}=\langle y\rangle^{1-|\alpha|}$, so
\[
\big|\partial_y^\alpha\langle y\rangle\big|\le C_\alpha\,\langle y\rangle^{1-|\alpha|} .
\]
In words, $\langle y\rangle$ behaves like a function homogeneous of degree one.

\paragraph{The bound.} Differentiating under the integral sign and using
$\langle y\rangle=H\ge1$, we obtain for $|\alpha|\ge1$
\[
\big|\partial_\eta^\alpha G(t,\eta)\big|
\le C_\alpha\int_{1/t}^1\sigma^{\,\ell_k+\sum_{j\ne k}\alpha_j(\ell_j-\ell_k)}\,
H(\sigma,\eta)^{1-|\alpha|}\,d\sigma
\le C_\alpha\int_{1/t}^1\sigma^{\,\ell_k+\sum_{j\ne k}\alpha_j(\ell_j-\ell_k)}\,d\sigma .
\]
The last integral stays bounded as $t\to\infty$ only if the exponent of $\sigma$ is
larger than $-1$. When some $\ell_j<\ell_k$, this fails for large $\alpha_j$. We
look at this more closely.

\subsection{Good, bad and oscillatory directions}

For $j\ne k$ we call the direction $j$
\begin{itemize}
	\item \emph{good} if $\ell_j\ge\ell_k$, and \emph{bad} if $\ell_j<\ell_k$;
	\item \emph{oscillatory} if $2\ell_j-\ell_k+1>0$.
\end{itemize}
We write $\mathcal O$ for the set of oscillatory directions. Every good direction is
oscillatory, since then $2\ell_j-\ell_k+1\ge\ell_k+1>0$. So a non-oscillatory
direction is bad. We define the weights
\[
v_j(\eta)=1\ \ (j\text{ good}),\qquad v_j(\eta)=|\eta_j|\ \ (j\text{ bad}).
\]

\paragraph{Why the direct bound fails.} High derivatives in a bad direction blow up
at $\eta_j=0$. Indeed, set all $\eta_m$ with $m\ne j$ to zero and expand
$H=(1+\sigma^{2(\ell_j-\ell_k)}\eta_j^2)^{1/2}$ in powers of $\eta_j$. The
coefficient of $\eta_j^{2N}$ is $\binom{1/2}{N}\sigma^{2N(\ell_j-\ell_k)}$, so
\[
\partial_{\eta_j}^{2N}G(t,0)=(2N)!\binom{1/2}{N}\int_{1/t}^1\sigma^{\,\ell_k+2N(\ell_j-\ell_k)}\,d\sigma .
\]
Since $\ell_j<\ell_k$, the exponent is below $-1$ for $N$ large, and the integral
tends to infinity as $t\to\infty$. The reason is a crossover: near $\eta_j=0$ the
coordinate $j$ dominates $H$ at early times $\sigma\sim1/t$, where
$\sigma^{\ell_j-\ell_k}$ is huge. Away from $\eta_j=0$ this does not happen, and the
weight $v_j=|\eta_j|$ measures how far away we are.

\subsection{The weighted bound}

\paragraph{Derivatives of $H$, weighted.} Write $\mu_m=\sigma^{\ell_m-\ell_k}$, so
that $y_m=\mu_m\eta_m$. The chain rule and the bound for $\langle y\rangle$ give
\[
\big|\partial_\eta^\alpha H\big|\le C_\alpha\,\mu^\alpha\,H^{1-|\alpha|}
=C_\alpha\,H\prod_{m\ne k}\Big(\frac{\mu_m}{H}\Big)^{\alpha_m},
\qquad\mu^\alpha=\prod_{m\ne k}\mu_m^{\alpha_m}.
\]
Two simple facts control each factor $\mu_m/H$:
\begin{itemize}
	\item if $m$ is good, then $\mu_m\le1\le H$, so $\mu_m/H\le1$;
	\item if $m$ is bad, then $H\ge\mu_m|\eta_m|$, so $\mu_m/H\le|\eta_m|^{-1}$.
\end{itemize}
In both cases $\mu_m/H\le v_m^{-1}$.

\begin{lemma}\label{lem:weighted}
	Let $|\alpha|\ge2$ with $\alpha_m=0$ for $m\notin\mathcal O$, and choose
	$i,j\in\mathcal O$ (possibly $i=j$) such that $\alpha'=\alpha-\delta_i-\delta_j$
	has nonnegative entries. Then for $t\ge2$,
	\[
	\big|\partial_\eta^\alpha G(t,\eta)\big|\le C_\alpha\prod_{m\ne k}v_m(\eta)^{-\alpha'_m}.
	\]
\end{lemma}

In words: two derivatives in oscillatory directions are free, and each further
derivative in a bad direction costs a factor $1/|\eta_m|$. Further derivatives in
good directions cost nothing.

\begin{proof}
	Since $|\alpha|=|\alpha'|+2$, we can split the bound for $\partial_\eta^\alpha H$ as
	\[
	\mu^\alpha H^{1-|\alpha|}=\frac{\mu_i\mu_j}{H}\,\prod_{m\ne k}\Big(\frac{\mu_m}{H}\Big)^{\alpha'_m}
	\le\frac{\mu_i\mu_j}{H}\,\prod_{m\ne k}v_m^{-\alpha'_m}.
	\]
	The product on the right does not depend on $\sigma$. It remains to integrate the
	first factor against $\sigma^{\ell_k}d\sigma$. Using $H\ge1$ and
	$\mu_i\mu_j\le\frac12(\mu_i^2+\mu_j^2)$,
	\[
	\int_{1/t}^1\sigma^{\ell_k}\,\frac{\mu_i\mu_j}{H}\,d\sigma
	\le\frac12\int_0^1\big(\sigma^{2\ell_i-\ell_k}+\sigma^{2\ell_j-\ell_k}\big)\,d\sigma<\infty,
	\]
	because $i,j\in\mathcal O$ means $2\ell_i-\ell_k>-1$ and $2\ell_j-\ell_k>-1$.
\end{proof}

\subsection{What the weights suggest}

The natural scale of a stationary phase integral with parameter $\lambda$ is
$\lambda^{-1/2}$. Zoom to this scale by setting $\omega=\lambda^{1/2}\eta$. Then
\[
\partial_\omega^\alpha(\lambda G)=\lambda^{1-|\alpha|/2}\,\partial_\eta^\alpha G
=\lambda^{-|\alpha'|/2}\,\partial_\eta^\alpha G ,
\]
and Lemma~\ref{lem:weighted} gives
\[
\big|\partial_\omega^\alpha(\lambda G)\big|\le C_\alpha\prod_{m\ne k}\big(\lambda^{1/2}v_m\big)^{-\alpha'_m}.
\]
For good $m$ we have $\lambda^{1/2}v_m=\lambda^{1/2}\ge1$, and for bad $m$ we have
$\lambda^{1/2}v_m=|\omega_m|$. So all $\omega$-derivatives of $\lambda G$ are
bounded as long as $|\omega_m|\ge1$ in the bad directions, that is,
$|\eta_m|\ge\lambda^{-1/2}$. This suggests the following splitting of the $\eta$
integral, which is carried out in Section~\ref{sec:proof}:
\begin{itemize}
	\item on the strips $|\eta_m|\le\lambda^{-1/2}$, $m$ bad, we integrate trivially;
	a strip has width $\lambda^{-1/2}$, which is exactly the gain oscillation would
	give;
	\item on the rest, the phase $\lambda G$ is convex with bounded derivatives in
	$\omega$, so the $\omega$ integral is bounded and each oscillatory direction gives
	the factor $\lambda^{-1/2}$;
	\item directions outside $\mathcal O$ are never differentiated and are integrated
	trivially.
\end{itemize}

\section{The stationary phase theorem}\label{sec:spt}

\subsection{Smooth sectors}

The stationary phase estimate below needs a smooth amplitude, and the indicator of
a sector is not smooth. So we replace the sharp sectors by a smooth partition. Fix
smooth $\rho_1,\dots,\rho_d$ on $\mathbb S^{d-1}$ with $\sum_k\rho_k=1$ and
$\rho_k(\theta)=0$ unless $|\theta_k|\ge\frac1{2\sqrt d}$. Such a partition exists
since $\max_k|\theta_k|\ge1/\sqrt d$ on $\mathbb S^{d-1}$. We insert
$\rho_k(A_t\xi/|A_t\xi|)$ into $I_k$ in place of the indicator of sector $k$. The
$\eta$ integral then runs over $\R^{d-1}$, with amplitude
\[
b(\xi_k,\eta)=(1+|\eta|^2)^{-1/2}\,\tilde\chi(\xi_k,\eta)\,
\rho_k\Big(\frac{\xi_k}{|\xi_k|}\,\frac{e_k+\sum_{j\ne k}\eta_je_j}{(1+|\eta|^2)^{1/2}}\Big),
\]
where
$\tilde\chi(\xi_k,\eta)=\chi\big(\xi_ke_k+\sum_{j\ne k}t^{\ell_k-\ell_j}\xi_k\eta_j\,e_j\big)$
is the cutoff in the new variables. The $k$th component of the argument of $\rho_k$
is $\pm(1+|\eta|^2)^{-1/2}$, so $b(\xi_k,\cdot)$ is smooth and supported in
$|\eta|\le2\sqrt d$. The bounds of Section~\ref{sec:sector-integral} on the support
remain true with other constants: $|\xi_k|\le2$ and $\lambda\ge c\,t^{1+\ell_{\min}}$.

\subsection{The theorem}

For $b\in C_0^\infty(\R^n)$ with support $K$, let $K_1$ be the set of points at sup
distance at most $1$ from $K$, and set
\[
M_m=\sum_{2\le|\alpha|\le m}\ \sup_{K_1}|\partial^\alpha\Phi|,\qquad
N_m=\sum_{|\alpha|\le m}\ \sup_K|\partial^\alpha b|,\qquad
a_0=\inf_{K_1}\big|\det\nabla^2\Phi\big| .
\]

\begin{theorem}[Alazard, Burq, Zuily, Theorem 3]\label{thm:ABZ}
	Let $\Phi$ be real and smooth near $K_1$, with $a_0>0$ and $\nabla\Phi$ injective
	on $K_1$. Then for all $\lambda\ge1$,
	\[
	\Big|\int_{\R^n}e^{i\lambda\Phi(\eta)}\,b(\eta)\,d\eta\Big|
	\le\frac{C_n}{a_0}\,\big(1+M_{n+2}\big)^{n/2}\,N_{n+1}\,\lambda^{-n/2},
	\]
	where $C_n$ depends only on $n$.
\end{theorem}

Two features of this bound matter for us. It involves no volume of $K$, so a large
support costs nothing. And it sees $\Phi$ only through derivatives of order at
least two, so the linear part of $\Phi$, which carries $x$, plays no role.

\subsection{A first application, and what it loses}

We apply Theorem~\ref{thm:ABZ} with $n=d-1$, for fixed $t$, $x$, $\xi_k$. Then
$K_1\subset[-R,R]^{d-1}$ with $R=2\sqrt d+1$. We check the hypotheses one by one.
\begin{itemize}
	\item \emph{Injectivity.} $\nabla_\eta^2\Phi$ is negative definite on $\R^{d-1}$,
	so for $\eta\ne\eta'$,
	\[
	\big\langle\nabla\Phi(\eta)-\nabla\Phi(\eta'),\eta-\eta'\big\rangle
	=\int_0^1\big\langle\nabla_\eta^2\Phi\big(\eta'+s(\eta-\eta')\big)(\eta-\eta'),\eta-\eta'\big\rangle\,ds<0 .
	\]
	Hence $\nabla\Phi$ is injective on all of $\R^{d-1}$, and no localization is
	needed.
	\item \emph{$a_0$.} By the lower bound of Section~\ref{sec:hessian}, all
	eigenvalues of $-\nabla_\eta^2\Phi$ are at least $c_R$ on $K_1$, so
	$a_0\ge c_R^{\,d-1}$.
	\item \emph{$M_{d+1}$.} For fixed $t$ the phase is smooth, so $M_{d+1}$ is finite.
	But Section~\ref{sec:higher} controls it uniformly in $t$ only in part: the direct
	bound grows with $t$ in bad directions, and Lemma~\ref{lem:weighted} degenerates
	near the hyperplanes $\eta_m=0$, $m$ bad.
	\item \emph{$N_d$.} The factors $(1+|\eta|^2)^{-1/2}$ and $\rho_k(\cdots)$ are
	smooth in $\eta$ and independent of $t$. Each $\partial_{\eta_j}$ falling on
	$\tilde\chi$ produces a factor $t^{\ell_k-\ell_j}\xi_k$, and $|\xi_k|\le2$, so
	\[
	N_d\le C\,t^{\,d(\ell_k-\ell_{\min})} .
	\]
	\item \emph{$\lambda\ge1$.} On the support of $b$,
	$\lambda\ge c\,t^{1+\ell_{\min}}$, which is $\ge1$ for $t$ large.
\end{itemize}

\paragraph{Conclusion of the first application.} For $t$ large,
\[
\Big|\int_{\R^{d-1}}e^{i\lambda\Phi(\eta)}\,b(\xi_k,\eta)\,d\eta\Big|
\le\frac{C}{a_0}\,\big(1+M_{d+1}\big)^{\frac{d-1}{2}}\,t^{\,d(\ell_k-\ell_{\min})}\,
\big(t^{\ell_k+1}|\xi_k|\big)^{-\frac{d-1}{2}} .
\]
None of $a_0$, $M_{d+1}$, $N_d$ depends on $x$, so the bound is uniform in $x$. But
it is not satisfactory in $t$: the factor $t^{d(\ell_k-\ell_{\min})}$ may cancel the
decay $\lambda^{-(d-1)/2}$, and $M_{d+1}$ is not under control uniformly in $t$.

\paragraph{Where the loss comes from.} Both problems have the same origin. The
quantities $M$ and $N$ measure derivatives at unit scale in $\eta$, but the
stationary phase integral lives at scale $\lambda^{-1/2}$. At that scale the picture
is different. A derivative of $\tilde\chi$ in direction $j$ costs
$t^{\ell_k-\ell_j}|\xi_k|$ per unit of $\eta_j$, hence
\[
\lambda^{-1/2}\,t^{\ell_k-\ell_j}|\xi_k|=t^{-(2\ell_j-\ell_k+1)/2}\,|\xi_k|^{1/2}
\]
per unit of $\omega_j=\lambda^{1/2}\eta_j$, and this is bounded exactly when $j$ is
oscillatory. And the strips $|\eta_m|\le\lambda^{-1/2}$, where
Lemma~\ref{lem:weighted} degenerates, are so thin that they can be integrated
trivially. The next section turns this into a proof.

\section{Proof of the main estimate}\label{sec:proof}

\subsection{Three groups of directions}

For $j\ne k$ we distinguish
\begin{itemize}
	\item $\mathcal O$, the oscillatory directions: $2\ell_j-\ell_k+1>0$;
	\item $\mathcal T$, the non-oscillatory directions: $2\ell_j-\ell_k+1\le0$;
	\item $\mathcal B\subset\mathcal O$, the bad oscillatory directions:
	$\ell_j<\ell_k$ and $j\in\mathcal O$.
\end{itemize}
Thus $\mathcal O\setminus\mathcal B$ consists of the good directions, and every
direction in $\mathcal T$ is bad. The directions in $\mathcal O$ will be treated by
stationary phase, those in $\mathcal T$ by the size of their support.

\subsection{Statement}

Recall from Section~\ref{sec:spt} that $b(\xi_k,\cdot)$ is smooth and supported in
$|\eta|\le2\sqrt d$, and that $|\xi_k|\le2$ and $\lambda\ge c\,t^{1+\ell_{\min}}$
whenever $b(\xi_k,\cdot)$ is not identically zero.

\begin{theorem}[Main estimate]\label{thm:main}
	There are $t_0$ and $C$, depending only on $d$ and $\ell_1,\dots,\ell_d$, such
	that for $t\ge t_0$, all $x\in\R^d$ and all $\xi_k\ne0$,
	\[
	\Big|\int_{\R^{d-1}}e^{i\lambda\Phi(\eta)}\,b(\xi_k,\eta)\,d\eta\Big|
	\le C\,\lambda^{-\frac{d-1}{2}}\prod_{j\in\mathcal T}t^{\frac{2\ell_j-\ell_k+1}{2}}\,|\xi_k|^{-1/2},
	\qquad\lambda=t^{\ell_k+1}|\xi_k| .
	\]
\end{theorem}

Each factor in the product over $\mathcal T$ is at most $|\xi_k|^{-1/2}$, since
$2\ell_j-\ell_k+1\le0$ there. When $\mathcal T$ is empty the estimate is the full
stationary phase rate $\lambda^{-(d-1)/2}$, with no loss from the amplitude.

\begin{remark}
	In the $\mathcal T$ directions the gain is $\lambda^{-1/2}$ only up to a factor
	$|\xi_k|^{-1/2}$. This factor has to be handled in the outer $\xi_k$ integral of
	$I_k$.
\end{remark}

\subsection{Strategy}

Set $n=d-1$ and write $J$ for the integral in Theorem~\ref{thm:main}, with $t$, $x$,
$\xi_k$ fixed. If $b(\xi_k,\cdot)\equiv0$ there is nothing to prove, so we assume
$|\xi_k|\le2$ and $\lambda\ge c\,t^{1+\ell_{\min}}$, and we take $t_0$ so large that
$\lambda\ge1$. The proof has five steps.
\begin{enumerate}
	\item Zoom to $\omega=\lambda^{1/2}\eta$. This puts the decay $\lambda^{-n/2}$ in
	front and leaves an integral that only has to be bounded.
	\item In each bad oscillatory direction, split the integral into a strip
	$|\omega_m|\le3$ and its complement.
	\item On each piece, apply Theorem~\ref{thm:ABZ} with $\lambda=1$ in the
	oscillatory directions that are not in a strip. Lemma~\ref{lem:weighted} gives
	bounded phase derivatives there, and the oscillatory condition gives bounded
	amplitude derivatives.
	\item Integrate trivially in the remaining directions: strips have bounded width,
	and non-oscillatory directions have thin support.
	\item Add up the pieces.
\end{enumerate}

\subsection{Step 1: zoom}

Put $\eta=\lambda^{-1/2}\omega$. Then $d\eta=\lambda^{-n/2}d\omega$ and
\[
J=\lambda^{-n/2}\int_{\R^n}e^{i\tilde\Phi(\omega)}\,\tilde b(\omega)\,d\omega ,
\qquad
\tilde\Phi(\omega)=\lambda\,\Phi(\lambda^{-1/2}\omega),
\quad
\tilde b(\omega)=b(\xi_k,\lambda^{-1/2}\omega).
\]
By the chain rule,
\[
\partial_\omega^\alpha\tilde\Phi(\omega)=\lambda^{1-|\alpha|/2}\,(\partial_\eta^\alpha\Phi)(\lambda^{-1/2}\omega).
\]
For $|\alpha|=2$ the power of $\lambda$ is zero, so the Hessian of $\tilde\Phi$ in
$\omega$ equals the Hessian of $\Phi$ in $\eta$ at the corresponding point. All the
decay $\lambda^{-n/2}$ now sits in front. It remains to bound the $\omega$ integral
by a constant, which is Theorem~\ref{thm:ABZ} with $\lambda=1$.

\subsection{Step 2: strip cutoffs in the bad directions}

Take $\psi\in C_0^\infty(\R)$ with $\psi=1$ on $[-2,2]$ and $\psi=0$ outside
$[-3,3]$. For each $m\in\mathcal B$ write $1=\psi(\omega_m)+(1-\psi(\omega_m))$ and
expand the product over $m\in\mathcal B$. This splits $J$ into $2^{|\mathcal B|}$
pieces $J_S$, one for each subset $S\subset\mathcal B$ (the directions placed in
strips):
\[
J_S=\lambda^{-n/2}\int_{\R^n}e^{i\tilde\Phi}\,\tilde b\,
\prod_{m\in S}\psi(\omega_m)\prod_{m\in\mathcal B\setminus S}\big(1-\psi(\omega_m)\big)\,d\omega .
\]
In $J_S$ we call $A=\mathcal O\setminus S$ the \emph{active} directions. We
integrate first in $\omega_A$, with the \emph{frozen} variables
$\omega_{S\cup\mathcal T}$ held fixed, and then in the frozen variables.

\subsection{Step 3: stationary phase in the active variables}

Fix $S$ and the frozen variables $\omega_{S\cup\mathcal T}$, and consider the inner
integral in $\omega_A\in\R^{|A|}$. We apply Theorem~\ref{thm:ABZ} in dimension
$|A|$, with $\lambda=1$, with phase $\tilde\Phi$ viewed as a function of $\omega_A$,
and with amplitude $\tilde b$ times the cutoffs. The theorem gives the bound
$\frac{C}{a_0}(1+M)^{|A|/2}N$ with no factor of the volume of the support, so the
support of size $\lambda^{1/2}$ in $\omega$ costs nothing. We check the hypotheses.

\paragraph{(a) The determinant $a_0$.} The support of $\tilde b$ lies in
$|\eta|\le2\sqrt d$. Its sup-distance-$1$ neighborhood $\tilde K_1$ in $\omega$ is a
sup-distance-$\lambda^{-1/2}$ neighborhood in $\eta$, and $\lambda^{-1/2}\le1$, so
$\tilde K_1$ lies in the ball $|\eta|\le3\sqrt d$. There the lower bound of
Section~\ref{sec:hessian} gives $-\nabla_\eta^2\Phi\ge cI$, hence
$-\nabla_\omega^2\tilde\Phi\ge cI$ by Step~1. The Hessian in the active variables is
a principal submatrix, and a principal submatrix of a matrix $\ge cI$ is again
$\ge cI$. Hence $a_0\ge c^{|A|}$.

\paragraph{(b) Injectivity.} $\tilde\Phi$ is strictly concave in $\omega_A$ on all
of $\R^{|A|}$, so $\nabla_{\omega_A}\tilde\Phi$ is injective by the monotonicity
argument of Section~\ref{sec:spt}.

\paragraph{(c) The phase derivatives $\tilde M$.} Let $|\alpha|\ge2$ with $\alpha$
supported in $A\subset\mathcal O$. Then $\partial^\alpha\Phi=-\partial^\alpha G$,
and Step~1 together with Lemma~\ref{lem:weighted} gives
\[
|\partial_\omega^\alpha\tilde\Phi|=\lambda^{1-|\alpha|/2}|\partial_\eta^\alpha G|
\le C\prod_{m}\big(\lambda^{1/2}v_m\big)^{-\alpha'_m}.
\]
For good $m$, $\lambda^{1/2}v_m=\lambda^{1/2}\ge1$. For bad $m\in A$, that is
$m\in\mathcal B\setminus S$, we have $\lambda^{1/2}v_m=|\omega_m|$, and
$|\omega_m|\ge1$ on $\tilde K_1$: the factor $1-\psi(\omega_m)$ vanishes for
$|\omega_m|\le2$, and $\tilde K_1$ extends the support by sup distance $1$. So every
factor is at most $1$, and $\tilde M_{|A|+2}\le C$.

\paragraph{(d) The amplitude derivatives $\tilde N$.} One $\omega_j$ derivative is
$\lambda^{-1/2}\partial_{\eta_j}$. Falling on $(1+|\eta|^2)^{-1/2}$ or on $\rho_k$,
it costs at most $C\lambda^{-1/2}\le C$. Falling on $\tilde\chi$, it produces the
factor $t^{\ell_k-\ell_j}\xi_k$, so it costs
\[
\lambda^{-1/2}\,t^{\ell_k-\ell_j}|\xi_k|=t^{-(2\ell_j-\ell_k+1)/2}\,|\xi_k|^{1/2}
\le\sqrt2\qquad(j\in\mathcal O,\ |\xi_k|\le2).
\]
This is where the oscillatory condition is used: it makes the exponent of $t$
negative. Derivatives of the strip cutoffs are $O(1)$. Hence
$\tilde N_{|A|+1}\le C$.

\paragraph{Conclusion of Step 3.} For every $S$ and every value of the frozen
variables $\omega_{S\cup\mathcal T}$,
\[
\Big|\int_{\R^{|A|}}e^{i\tilde\Phi}\,\tilde b\,
\prod_{m\in S}\psi(\omega_m)\prod_{m\in\mathcal B\setminus S}\big(1-\psi(\omega_m)\big)\,d\omega_A\Big|\le C .
\]

\subsection{Step 4: trivial integration in the frozen variables}

It remains to integrate the constant $C$ over the frozen variables, which are
confined to a box.
\begin{itemize}
	\item $m\in S$. The cutoff $\psi(\omega_m)$ is supported in $|\omega_m|\le3$, so
	this costs a factor $6$.
	\item $j\in\mathcal T$. On the support of $\tilde\chi$ we have $|\xi_j|\le2$ with
	$\xi_j=t^{\ell_k-\ell_j}\xi_k\eta_j$, so $|\eta_j|\le2t^{\ell_j-\ell_k}/|\xi_k|$.
	In the variable $\omega_j=\lambda^{1/2}\eta_j$ this means $|\omega_j|\le W_j$ with
	\[
	W_j=\lambda^{1/2}\cdot\frac{2t^{\ell_j-\ell_k}}{|\xi_k|}
	=2\,t^{\frac{2\ell_j-\ell_k+1}{2}}\,|\xi_k|^{-1/2}
	\le2\,|\xi_k|^{-1/2}\qquad(j\in\mathcal T),
	\]
	so this costs a factor $2W_j$. Here the non-oscillatory condition
	$2\ell_j-\ell_k+1\le0$ is used: the support in $\omega_j$ is no wider than a
	constant times $|\xi_k|^{-1/2}$.
\end{itemize}

\subsection{Step 5: conclusion}

Combining Steps 3 and 4, each piece satisfies
\[
|J_S|\le C\,\lambda^{-n/2}\,6^{|S|}\prod_{j\in\mathcal T}2W_j
\le C\,\lambda^{-\frac{d-1}{2}}\prod_{j\in\mathcal T}t^{\frac{2\ell_j-\ell_k+1}{2}}\,|\xi_k|^{-1/2}.
\]
Summing over the $2^{|\mathcal B|}$ subsets $S$ proves Theorem~\ref{thm:main}. The
bound is uniform in $x$ because $x$ enters $\Phi$ only through its linear part,
which none of $a_0$, $\tilde M$, $\tilde N$ sees.

\paragraph{Summary.} After zooming to $\omega=\lambda^{1/2}\eta$, the oscillatory
condition $2\ell_j-\ell_k+1>0$ is exactly what keeps the amplitude smooth at scale
$1$. Every direction where it fails has support thinner than the stationary phase
scale, so it needs no oscillation. The bad oscillatory directions are handled by
cutting out a strip of unit width in $\omega$, where the weighted bounds
degenerate; integrating over the strip costs the same factor that oscillation would
gain.
